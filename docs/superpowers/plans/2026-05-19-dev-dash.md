export function AppRailItem({ to, label, icon: Icon }: AppRailItemProps) {
  return (
    <NavLink to={to} title={label} end>

          variant="ghost"
          size="icon"
          className={cn(
            "size-10",
            isActive &&
              "bg-primary text-primary-foreground hover:bg-primary/90 hover:text-primary-foreground",
          )}
          aria-label={label}
          aria-current={isActive ? "page" : undefined}
        >
          <Icon className="size-5" />
        </Button>
      )}
    </NavLink>
  );
}

- [ ] **Step 3: App.tsx 路由**

```tsx
import { BrowserRouter, Navigate, Route, Routes } from "react-router-dom";

import { AppShell } from "@/app/AppShell";
import { modules } from "@/app/modules";
import { HealthCheckPage } from "@/pages/HealthCheckPage";

export function App() {
  return (
    <BrowserRouter>
      <Routes>
        <Route element={<AppShell />}>
          <Route index element={<Navigate to="/env" replace />} />
          {modules.map((m) => {
            const Page = m.page;
            return <Route key={m.id} path={m.path} element={<Page title={m.label} />} />;
          })}
        </Route>
        <Route path="/debug/health" element={<HealthCheckPage />} />
        <Route path="*" element={<Navigate to="/env" replace />} />
      </Routes>
    </BrowserRouter>
  );
}
```

注意：`Route path={m.path}` 应使用不含前导斜杠的 path 段，或统一用 `path="env"` 且 `path` 在 modules 存 `env`。推荐 modules 改为：

```ts
path: "env",  // 非 "/env"
```

注册表与 NavLink 的 `to` 使用 `/env` 绝对路径；Route 使用 `path="env"`。

修正 `modules.ts`：

```ts
path: "env",   // Route segment
// NavLink: to={`/${m.path}`} 或在 modules 增加 fullPath: "/env"
```

**更简单做法：** modules 保留 `path: "/env"`，Route 使用：

```tsx
<Route path={m.path.replace(/^\//, "")} element={...} />
```

或 modules 双字段。计划采用 **routePath** + **navPath**：

```ts
export type DevDashModule = {
  id: DevDashModuleId;
  label: string;
  routePath: string; // "env"
  navPath: string; // "/env"
  icon: LucideIcon;
  page: ComponentType<{ title: string }>;
};
```

实施时二选一，保持 NavLink `to={navPath}` 与 `<Route path={routePath}>` 一致。
